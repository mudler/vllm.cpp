cmake_minimum_required(VERSION 3.20)
project(strix_matched_adapters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
foreach(required VLLM_INCLUDE VLLM_LIBRARY LLAMA_INCLUDE LLAMA_LIBRARY GGML_INCLUDE JSON_INCLUDE)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Supply bound ${required}; this build does not fetch or build dependencies")
  endif()
endforeach()
add_executable(strix-vllm-adapter native_adapter.cpp)
target_include_directories(strix-vllm-adapter PRIVATE "${VLLM_INCLUDE}" "${JSON_INCLUDE}")
target_link_libraries(strix-vllm-adapter PRIVATE "${VLLM_LIBRARY}" Threads::Threads)
add_executable(strix-llama-adapter native_adapter.cpp)
target_compile_definitions(strix-llama-adapter PRIVATE STRIX_LLAMA)
target_include_directories(strix-llama-adapter PRIVATE "${LLAMA_INCLUDE}" "${GGML_INCLUDE}" "${JSON_INCLUDE}")
target_link_libraries(strix-llama-adapter PRIVATE "${LLAMA_LIBRARY}" Threads::Threads)
